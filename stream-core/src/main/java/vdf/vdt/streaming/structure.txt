vdf-vdt-streaming/
├── pom.xml
├── src/main/resources/
│   └── config.dev.yml                         # Cấu hình Kafka brokers, topics, app configs
└── src/main/java/vdf/vdt/streaming/
    ├── config/
    │   └── ConfigLoader.java                  # Đọc cấu hình từ file YAML
    │
    ├── deserializer/
    │   ├── KafkaDataDeserializationSchema.java   # Đọc Kafka Header (source, version) + Payload thành RawKafkaEvent
    │   └── KafkaSchemaDeserializationSchema.java # (Tùy chọn) Đọc Schema payload/header từ topic schema
    │
    ├── model/
    │   ├── event/
    │   │   ├── RawKafkaEvent.java             # POJO chứa header metadata và raw payload byte[]
    │   │   └── GenericEvent.java              # POJO chuẩn chứa customer_id, event_time và Map fields đã parse & cast
    │   └── schema/
    │       ├── DataType.java                  # Enum: INT, LONG, FLOAT, DOUBLE, STRING, TIMESTAMP, BOOLEAN, OBJECT
    │       ├── FieldCategory.java             # Enum: dynamic_numeric, static_categorical, dynamic_categorical, static_numeric
    │       ├── SourceVersionKey.java          # Record: (source, version) làm định danh Schema
    │       ├── ColumnDefinition.java          # Chi tiết cột (fieldPath, dataType, category)
    │       └── TableSchema.java               # Lưu trữ schema hoàn chỉnh (Map<String, ColumnDefinition>)
    │
    ├── parser/
    │   ├── event/
    │   │   └── DynamicEventValidator.java     # Đệ quy parse JSON event và ép kiểu theo TableSchema
    │   └── schema/
    │       └── DynamicSchemaJsonParser.java   # Parse JSON từ topic schema thành TableSchema
    │
    ├── processor/
    │   └── DynamicEventBroadcastProcessor.java # BroadcastProcessFunction<RawKafkaEvent, byte[], GenericEvent>
    │
    ├── state/
    │   └── StateDescriptors.java              # Quản lý SCHEMA_BROADCAST_STATE
    │
    └── StreamingJob.java                      # Dựng Dataflow: Data Source -> Connect Schema Broadcast -> Parse -> KeyBy


[1. Schema Config JSON]
         │
         ▼
[DynamicSchemaJsonParser] ──creates──► [TableSchema] (Chứa Map<String, ColumnDefinition>)
                                             │
                                             │ (Lưu vào Schema Registry / Cache / Flink State)
                                             ▼
[2. Raw Event JSON] ───────► [DynamicEventValidator]
                                    │
                                    ├── Đọc TableSchema để lấy kiểu dữ liệu
                                    ├── Bóc tách, đệ quy flatten các trường lồng nhau
                                    └── Ép kiểu theo DataType
                                    │
                                    ▼
                             [GenericEvent] (Đưa vào Engine xử lý luồng)


[config-schema-topic] ──► Flink Kafka Source ──► (DynamicSchemaJsonParser) ──► BroadcastStream (MapStateDescriptor)
                                                                                         │
                                                                                         ▼ (Broadcast)
[event-topics (1, 2)] ──► Flink Kafka Source ──► KeyBy (customer/key_field) ──► KeyedBroadcastProcessFunction
                                                                                         │
                                                                                         ├── 1. Lookup Schema from State
                                                                                         ├── 2. DynamicEventValidator
                                                                                         └── 3. Sink to ClickHouse
